/* PROFILE UPLOAD */

function uploadProfile(event){

    const file=event.target.files[0];

    if(!file)return;

    if(file.size>5*1024*1024){
        alert("Rasm 5 MB dan kichik bo‘lishi kerak.");
        return;
    }

    const reader=new FileReader();

    reader.onload=function(){

        data.profile=reader.result;

        document.getElementById("adminProfilePreview").src=data.profile;

        saveData();
    };

    reader.readAsDataURL(file);
}


/* GALLERY UPLOAD */

function uploadGallery(event){

    const files=[...event.target.files];

    files.forEach(file=>{

        if(file.size>5*1024*1024)return;

        const reader=new FileReader();

        reader.onload=function(){

            data.gallery.push(reader.result);

            localStorage.setItem(
                "portfolioData",
                JSON.stringify(data)
            );

            renderGallery();
            renderAdmin();
        };

        reader.readAsDataURL(file);
    });
}

/* EXPORT */

function exportData(){

    const blob=new Blob(
        [JSON.stringify(data,null,2)],
        {type:"application/json"}
    );

    const url=URL.createObjectURL(blob);

    const a=document.createElement("a");
    a.href=url;
    a.download="portfolio-data.json";
    a.click();

    URL.revokeObjectURL(url);
}

let languages=["UZ","RU","EN"];
let langIndex=0;

function changeLang(){

    langIndex=(langIndex+1)%languages.length;

    const lang=languages[langIndex];

    document.querySelector(".header-tools .tool").textContent=lang;

    if(lang==="RU"){
        document.querySelector("#about .section-title h2").textContent="Обо мне";
        document.querySelector("#contact .section-title h2").textContent="Контакты";
    }

    if(lang==="EN"){
        document.querySelector("#about .section-title h2").textContent="About Me";
        document.querySelector("#contact .section-title h2").textContent="Contact";
    }

    if(lang==="UZ"){
        document.querySelector("#about .section-title h2").textContent="Men haqimda";
        document.querySelector("#contact .section-title h2").textContent="Bog‘lanish";
    }
}

/* =========================
   REVEAL
========================= */

function reveal(){

    document.querySelectorAll(".reveal").forEach(el=>{

        const top=el.getBoundingClientRect().top;

        if(top < window.innerHeight-80){
            el.classList.add("active");
        }

    });
}

window.addEventListener("scroll",reveal);


/* =========================
   SECURITY
========================= */

function escapeHTML(str){

    return String(str)
        .replaceAll("&","&amp;")
        .replaceAll("<","&lt;")
        .replaceAll(">","&gt;")
        .replaceAll('"',"&quot;")
        .replaceAll("'","&#039;");
}
